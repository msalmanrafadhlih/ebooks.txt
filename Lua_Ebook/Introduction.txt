OPERASI DASAR
1. Operasi Aritmetika

   > a, b = 10, 3
     print(a + b) -- pertambahan
     print(a - b) -- perkurangan
     print(a / c) -- pembagian
     print(a * b) -- perkalian
     print(a % b) -- Modulus(sisa)
     print(a ^ b) -- Eksponen
     Output ------- 13, 7, 3.33, 30, 1, 1000
	
2. Konkatenasi String
-> operator " .. " digunakan untuk konkatenasi string
-> operator " # " untuk mengambalikan panjang string/tabel
 
   > 1st_name, 2nd_name = "John", "Doe"
     fullname = 1st_name .. " " .. 2nd_name
     print(fullname) -- John Doe
     print(#fullname) -- 10(Total char)
