def sapa_dunia(nama):
    return f"Halo, {nama}! Selamat datang di Python."

# List comprehension yang ikonik
angka_genap = [x for x in range(10) if x % 2 == 0]

print(sapa_dunia("Programmer"))
print(f"Angka genap: {angka_genap}")
